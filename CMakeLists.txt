cmake_minimum_required(VERSION 3.28)
project(ChronicleLite VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHRONICLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(cmake/CompilerWarnings.cmake)
add_library(ChronicleRuntime src/binary.cpp src/archive.cpp src/schema.cpp src/serialization.cpp src/file.cpp src/migration.cpp)
target_include_directories(ChronicleRuntime PUBLIC include)
chronicle_options(ChronicleRuntime)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
add_library(ChronicleTools tools/cli/output.cpp tools/cli/commands.cpp)
target_include_directories(ChronicleTools PUBLIC tools/cli)
target_link_libraries(ChronicleTools PUBLIC ChronicleRuntime PRIVATE nlohmann_json::nlohmann_json)
chronicle_options(ChronicleTools)
add_executable(chronicle tools/cli/main.cpp)
target_link_libraries(chronicle PRIVATE ChronicleTools)
chronicle_options(chronicle)
add_executable(scene_demo examples/scene_demo/main.cpp)
target_link_libraries(scene_demo PRIVATE ChronicleRuntime)
chronicle_options(scene_demo)
add_executable(type_binding_demo examples/type_binding_demo/main.cpp)
target_link_libraries(type_binding_demo PRIVATE ChronicleTools)
chronicle_options(type_binding_demo)
add_executable(chronicle_benchmark benchmarks/main.cpp benchmarks/allocation_tracking.cpp)
target_link_libraries(chronicle_benchmark PRIVATE ChronicleRuntime)
chronicle_options(chronicle_benchmark)
include(CTest)
if(BUILD_TESTING)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926)
    FetchContent_MakeAvailable(googletest)
    # Apple Clang 21 warns on GoogleTest's widening char8_t printer.
    # Restrict this upstream warning workaround to the dependency itself.
    if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 21)
        target_compile_options(gtest PRIVATE -Wno-character-conversion)
    endif()
    add_executable(chronicle_tests tests/binary_tests.cpp tests/serialization_tests.cpp tests/file_tests.cpp tests/cli_tests.cpp tests/binding_tests.cpp tests/migration_v3_tests.cpp)
    target_link_libraries(chronicle_tests PRIVATE ChronicleTools GTest::gtest_main nlohmann_json::nlohmann_json)
    target_compile_definitions(chronicle_tests PRIVATE CHRONICLE_GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/golden")
    chronicle_options(chronicle_tests)
    add_test(NAME benchmark_allocation_tracking COMMAND chronicle_benchmark --check-allocations)
    include(GoogleTest)
    gtest_discover_tests(chronicle_tests)
    add_test(NAME cli_validate_v1 COMMAND chronicle validate ${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/scene_v1.bin)
    add_test(NAME cli_validate_v3 COMMAND chronicle validate ${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/scene_v3.bin)
    add_test(NAME cli_validate_v2 COMMAND chronicle validate ${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/scene_v2.bin)
    add_test(NAME demo_save COMMAND scene_demo save ${CMAKE_CURRENT_BINARY_DIR}/demo.bin)
    add_test(NAME demo_load COMMAND scene_demo load ${CMAKE_CURRENT_BINARY_DIR}/demo.bin)
    set_tests_properties(demo_load PROPERTIES DEPENDS demo_save)
    add_test(NAME binding_demo_save COMMAND type_binding_demo save ${CMAKE_CURRENT_BINARY_DIR}/settings.bin)
    add_test(NAME binding_demo_load COMMAND type_binding_demo load ${CMAKE_CURRENT_BINARY_DIR}/settings.bin)
    add_test(NAME binding_demo_inspect COMMAND type_binding_demo inspect ${CMAKE_CURRENT_BINARY_DIR}/settings.bin)
    set_tests_properties(binding_demo_load binding_demo_inspect PROPERTIES DEPENDS binding_demo_save)
    add_test(NAME cli_workflow COMMAND ${CMAKE_COMMAND}
        -DCLI=$<TARGET_FILE:chronicle>
        -DGOLDEN_DIR=${CMAKE_CURRENT_SOURCE_DIR}/tests/golden
        -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-workflow
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_workflow.cmake)
endif()
